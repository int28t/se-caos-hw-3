cmake_minimum_required(VERSION 3.20)
project(crossroad LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "ASan and UBSan" OFF)
add_compile_definitions(_POSIX_C_SOURCE=200809L)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(ENABLE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(crossroad STATIC src/config.c src/io.c src/model.c src/roles.c src/ipc.c src/ipc_shm.c src/ipc_fifo.c src/runtime.c)
target_include_directories(crossroad PUBLIC src)
find_package(Threads REQUIRED)
target_link_libraries(crossroad PUBLIC Threads::Threads)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(crossroad PUBLIC rt)
endif()
foreach(backend named unnamed shm fifo)
  string(TOUPPER ${backend} mode)
  add_executable(crossroad_${backend} src/main.c)
  target_compile_definitions(crossroad_${backend} PRIVATE BACKEND=${mode})
  target_link_libraries(crossroad_${backend} PRIVATE crossroad)
endforeach()
set(kind 1)
foreach(role controller direction participants)
  add_executable(crossroad_${role} src/main.c)
  target_compile_definitions(crossroad_${role} PRIVATE ROLE_KIND=${kind})
  target_link_libraries(crossroad_${role} PRIVATE crossroad)
  math(EXPR kind "${kind} + 1")
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_test tests/test_core.c)
  target_link_libraries(core_test PRIVATE crossroad)
  find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
  add_test(NAME scenarios COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run.py --build-dir ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(scenarios PROPERTIES TIMEOUT 180)
  add_test(NAME core COMMAND core_test ${CMAKE_CURRENT_SOURCE_DIR}/data/minimal.conf)
endif()
